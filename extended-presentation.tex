\begin{frame}{2. Question, claim, and non-claim}
\begin{block}{Research question}Does agentic AI expand the languages in which a developer can ship working code, beyond accelerating work in familiar languages?\end{block}
\vspace{.5em}
\textbf{Claim:} the developer's observed \alert{production frontier} expands.
\vspace{.5em}

\textbf{Non-claim:} the developer learns to code unassisted in the new language. Delegated output remains joint production: the human selects, specifies, verifies, and ships.
\vspace{.5em}

\textbf{Unit:} developer $i$ × language $k$ × month $t$.
\end{frame}

\begin{frame}{3. Three capabilities}
\begin{columns}[T,onlytextwidth]
\begin{column}{.31\textwidth}\begin{block}{Solo $S$}Expected human match $s\mu$, activation cost $b$, and CARA risk penalty from match uncertainty.\end{block}\end{column}
\begin{column}{.31\textwidth}\begin{block}{Augmentation $C$}Adds $\gamma s-r_C$: suggestions are useful only with language-specific foothold $s$.\end{block}\end{column}
\begin{column}{.31\textwidth}\begin{block}{Delegation $D$}Agent executes share $\lambda$; output uses general ability $a$, capability $z(A)$, verification cost $\kappa$, and agent-error risk.\end{block}\end{column}
\end{columns}
\vspace{.8em}
\centering\large One modelling choice drives the result: \alert{AI supplies delegated execution, not merely a substitutable productivity input.}
\end{frame}

\begin{frame}{4. The developer's choice is not effort}
\[
M^1=\{S,C\},\qquad M^2=\{S,C,D\},
\]
\[
V^g_{ikt}=\max_{m\in M^g}V^m_{ikt},\qquad
Z^g_{ikt}=\mathbf 1\{V^g_{ikt}\ge0\},\qquad N^g_{it}=\sum_k Z^g_{ikt}.
\]
\begin{block}{Economic decision}
Given an opportunity, choose the mode with the highest certainty-equivalent surplus, then enter only if that surplus is nonnegative.
\end{block}
\textbf{Audit point:} there is no endogenous continuous effort variable. An “agent effort problem” belongs to a different paper/model.
\end{frame}

\begin{frame}{5. Thresholds and the delegation advantage}
\small
\[
T^S=b-s\mu+\frac{\rho s^2}{2\pi},\qquad
T^1=T^S-\max\{0,\gamma s-r_C\}.
\]
For unfamiliar $k$, Assumption 1 imposes $\gamma s-r_C\le0$, so $T^1=T^S$. Define
\[
B\equiv T^S-T^D
=\underbrace{\lambda[az(A)-s\mu]}_{\text{execution substitution}}
-\underbrace{\kappa(a,s)-r_D}_{\text{verification + compute}}
+\underbrace{\frac{\rho}{2}\left[\frac{(2\lambda-\lambda^2)s^2}{\pi}-\sigma_D^2\right]}_{\text{risk substitution}}.
\]
\begin{alertblock}{Condition}Delegation changes entry only when $B>0$. Better verification technology can enlarge the band; it does not guarantee a positive band everywhere.\end{alertblock}
\end{frame}

\begin{frame}{6. Propositions 1--2}
\textbf{Proposition 1 (weak frontier expansion).}
\[
M^1\subset M^2\Longrightarrow Z^2_{ikt}\ge Z^1_{ikt}\Longrightarrow N^2_{it}\ge N^1_{it}.
\]
The agent can always be ignored; strict expansion is not automatic.
\vspace{.8em}

\textbf{Proposition 2 (activation band).} For unfamiliar $k$, Assumption 1 and $B>0$ imply
\[
\boxed{Z^2_{ikt}-Z^1_{ikt}=\mathbf 1\{T^D_{ikt}\le\omega_{ikt}<T^1_{ikt}\}}.
\]
With continuous conditional CDF $F$, activation probability is $F(T^1)-F(T^D)$.
\end{frame}

\begin{frame}{7. Proposition 3: stock and flow}
If constant first-use hazards satisfy $p^2_{ik}\ge p^1_{ik}$, then
\[
\Delta C_i(h)=\sum_{k\in U_i}\left[(1-p^1_{ik})^{h+1}-(1-p^2_{ik})^{h+1}\right]\ge0.
\]
\begin{columns}[T,onlytextwidth]
\begin{column}{.48\textwidth}\begin{block}{Closed frontier}If $p^1=0<p^2$, the cumulative gap grows strictly and concavely over the observed horizon.\end{block}\end{column}
\begin{column}{.48\textwidth}\begin{alertblock}{Endpoint/domain caveat}With $p^1>0$, monotonic growth needs the paper's additional “no catch-up” inequality; $p^2\ge p^1$ alone gives nonnegativity, not monotonicity.\end{alertblock}\end{column}
\end{columns}
\end{frame}

\begin{frame}{9. Empirical strategy}
\begin{block}{Estimator}
Callaway--Sant'Anna group-time ATT, doubly robust, not-yet-treated comparisons, one-month anticipation, varying base periods, 1,000 developer-clustered multiplier-bootstrap draws.
\end{block}
\textbf{Four frontier outcomes:} monthly distinct languages, newly used languages, cumulative languages, and Shannon entropy. Repositories and commits are diagnostics.
\vspace{.6em}
\begin{alertblock}{Interpretation discipline}
Conditional parallel trends handles staggered timing and observed pre-period differences. It does not handle a new-project shock that both induces Claude adoption and changes languages.
\end{alertblock}

\begin{frame}[fragile]{14. Proposition 3: paper $\rightarrow$ Lean $\rightarrow$ proof}
\scriptsize
\textbf{Paper statement.} If $0\le p^1_k\le p^2_k\le1$, then
$\Delta C(s)=\sum_{k\in U}[(1-p^1_k)^{s+1}-(1-p^2_k)^{s+1}]\ge0$; when
$U\ne\varnothing$ and $p^1_k=0<p^2_k<1$, the gap is strictly increasing and concave.

\textbf{Lean translation and proof fragment.}
\begin{lstlisting}[language={},literate={₀}{{\ensuremath{_0}}}1 {₁}{{\ensuremath{_1}}}1 {₂}{{\ensuremath{_2}}}1 {≤}{{\ensuremath{\leq}}}1 {≥}{{\ensuremath{\geq}}}1 {·}{{\textbullet}}1 {⟨}{{\ensuremath{\langle}}}1 {⟩}{{\ensuremath{\rangle}}}1 {Δ}{{\ensuremath{\Delta}}}1]
theorem dynamic_cumulative_language_effect :
  dynamic_cumulative_language_effectSpec := by
  intro Language _ unfamiliar p₁ p₂ hHazards
  dsimp; constructor
  · intro s                          -- weak gap: ΔC(s) ≥ 0
    apply Finset.sum_nonneg; intro k hk
    have hkB := hHazards k hk
    have h0  : 0 ≤ 1 - p₂ k := by linarith
    have hle : 1 - p₂ k ≤ 1 - p₁ k := by linarith
    have := pow_le_pow_left₀ h0 hle (s + 1)
    linarith
  · rintro ⟨hNe, hClosed⟩            -- closed frontier p₁=0<p₂:
    -- strictMono_nat_of_lt_succ + pow_lt_pow_right_of_lt_one₀ => strict growth + concavity
\end{lstlisting}
\textbf{Plain English.} A higher first-use hazard lowers every survival
probability $(1-p)^{s+1}$; summing those nonnegative differences gives $\Delta C(s)\ge0$.
Lean checks this \emph{exactly} under the paper's stated hypothesis $p^2\ge p^1$; the
strict-increase and concavity half is proved on the closed-frontier benchmark $p^1=0<p^2$,
precisely the domain the proposition conditions on---no extra assumption is smuggled in.

\begin{frame}{15. Verdict}
\Large
The model cleanly shows how delegated execution can expand a \alert{production} frontier through a lower entry threshold.
\vspace{.8em}

\normalsize
The data show the predicted activation spike, accumulation pattern, and specialist headroom signature. They do not causally separate delegation from project-driven adoption, and they do not show skill acquisition.
